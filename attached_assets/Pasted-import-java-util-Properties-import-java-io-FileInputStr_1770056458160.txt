// Fix output directory for Flutter/Shorebird compatibility
android.applicationVariants.all {
    val variant = this
    outputs.all {
        val outputDir = File(project.rootProject.projectDir.parentFile, "build/app/outputs/bundle/${variant.buildType.name}")
        outputDir.mkdirs()
    }
}

// Copy AAB to expected location after bundle task
tasks.whenTaskAdded {
    if (name == "bundleRelease") {
        doLast {
            val sourceAab = file("${layout.buildDirectory.get()}/outputs/bundle/release/app-release.aab")
            val targetDir = File(project.rootProject.projectDir.parentFile, "build/app/outputs/bundle/release")
            targetDir.mkdirs()
            if (sourceAab.exists()) {
                sourceAab.copyTo(File(targetDir, "app-release.aab"), overwrite = true)
                println("Copied AAB to: ${targetDir}/app-release.aab")
            }
        }
    }
    if (name == "assembleRelease") {
        doLast {
            val sourceApk = file("${layout.buildDirectory.get()}/outputs/apk/release/app-release.apk")
            val targetDir = File(project.rootProject.projectDir.parentFile, "build/app/outputs/flutter-apk")
            targetDir.mkdirs()
            if (sourceApk.exists()) {
                sourceApk.copyTo(File(targetDir, "app-release.apk"), overwrite = true)
                println("Copied APK to: ${targetDir}/app-release.apk")
            }
        }
    }
}
