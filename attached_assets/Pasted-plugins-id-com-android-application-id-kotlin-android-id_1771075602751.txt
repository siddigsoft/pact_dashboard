// Register task first
tasks.register('fixGeneratedPluginRegistrant') {
    doLast {
        def registrantFile = file("src/main/java/io/flutter/plugins/GeneratedPluginRegistrant.java")
        if (registrantFile.exists()) {
            def content = registrantFile.text
            if (!content.contains("import android.util.Log;") && content.contains("import io.flutter.Log;")) {
                content = content.replace("import io.flutter.Log;", "import android.util.Log;")
                registrantFile.text = content
                println("✓ Fixed GeneratedPluginRegistrant imports")
            }
        }
    }
}

// Make preBuild depend on the fix task
preBuild.dependsOn fixGeneratedPluginRegistrant
