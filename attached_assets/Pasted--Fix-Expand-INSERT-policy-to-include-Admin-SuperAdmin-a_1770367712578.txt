-- Fix: Expand INSERT policy to include Admin, SuperAdmin, and Supervisor roles
-- Previously only FOM, Coordinator, and CountryDirector could create submissions
-- This prevented admins and supervisors from submitting operational costs

-- Step 1: Drop the old restrictive INSERT policy
DROP POLICY IF EXISTS "Authorized roles can create operational cost submissions" ON operational_cost_submissions;

-- Step 2: Create new expanded INSERT policy
CREATE POLICY "Authorized roles can create operational cost submissions"
  ON operational_cost_submissions FOR INSERT
  WITH CHECK (
    auth.uid() = submitted_by
    AND EXISTS (
      SELECT 1 FROM profiles p
      WHERE p.id = auth.uid()
      AND p.role IN (
        'Field Operation Manager (FOM)', 
        'Coordinator', 'coordinator',
        'CountryDirector',
        'admin', 'SuperAdmin',
        'hubSupervisor', 'supervisor'
      )
    )
  );

-- Step 3: Update expense_category CHECK constraint to match frontend categories
ALTER TABLE operational_cost_submissions DROP CONSTRAINT IF EXISTS operational_cost_submissions_expense_category_check;

ALTER TABLE operational_cost_submissions ADD CONSTRAINT operational_cost_submissions_expense_category_check
  CHECK (expense_category IN (
    'permits', 'incentives', 'communications', 'training', 
    'general_transport', 'equipment', 'printing', 'meetings', 'other'
  ));

-- Step 4: Rename columns if they use old naming convention (reviewed -> approved)
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name = 'operational_cost_submissions' AND column_name = 'tier1_reviewed_by') THEN
    ALTER TABLE operational_cost_submissions RENAME COLUMN tier1_reviewed_by TO tier1_approved_by;
  END IF;
  IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name = 'operational_cost_submissions' AND column_name = 'tier1_reviewed_at') THEN
    ALTER TABLE operational_cost_submissions RENAME COLUMN tier1_reviewed_at TO tier1_approved_at;
  END IF;
  IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name = 'operational_cost_submissions' AND column_name = 'tier2_reviewed_by') THEN
    ALTER TABLE operational_cost_submissions RENAME COLUMN tier2_reviewed_by TO tier2_approved_by;
  END IF;
  IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name = 'operational_cost_submissions' AND column_name = 'tier2_reviewed_at') THEN
    ALTER TABLE operational_cost_submissions RENAME COLUMN tier2_reviewed_at TO tier2_approved_at;
  END IF;
  
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name = 'operational_cost_submissions' AND column_name = 'rejection_reason') THEN
    ALTER TABLE operational_cost_submissions ADD COLUMN rejection_reason TEXT;
  END IF;
END $$;

-- Step 5: Fix the Supervisors UPDATE policy to use correct column name
DROP POLICY IF EXISTS "Supervisors can update tier1 for hub submissions" ON operational_cost_submissions;

CREATE POLICY "Supervisors can update tier1 for hub submissions"
  ON operational_cost_submissions FOR UPDATE
  USING (
    tier1_status = 'pending'
    AND EXISTS (
      SELECT 1 FROM profiles p
      WHERE p.id = auth.uid()
      AND p.hub_id = operational_cost_submissions.hub_id
      AND p.role IN ('hubSupervisor', 'supervisor', 'Field Operation Manager (FOM)')
    )
  )
  WITH CHECK (
    tier2_status = 'pending'
    AND tier2_approved_by IS NULL
    AND wallet_transaction_id IS NULL
    AND paid_at IS NULL
  );
